Sub RemoveSlideNumberTextBoxes()
    Dim sld As Slide
    Dim shp As Shape
    
    For Each sld In ActivePresentation.Slides
        For Each shp In sld.Shapes
            If shp.HasTextFrame Then
                If shp.TextFrame.HasText Then
                    If shp.TextFrame.TextRange.Text Like "Slide *" Then
                        shp.Delete
                    End If
                End If
            End If
        Next shp
    Next sld
End Sub
