Sub DeleteQuestionGroupsFromAllSlides()
    Dim sld As Slide
    Dim shp As Shape
    Dim subShp As Shape
    Dim i As Long
    Dim j As Long
    Dim found As Boolean

    For Each sld In ActivePresentation.Slides
        ' Loop backwards to avoid skipping after deletion
        For i = sld.Shapes.Count To 1 Step -1
            Set shp = sld.Shapes(i)
            If shp.Type = msoGroup Then
                found = False
                ' Check each shape in the group
                For j = 1 To shp.GroupItems.Count
                    Set subShp = shp.GroupItems(j)
                    If subShp.HasTextFrame Then
                        If subShp.TextFrame.HasText Then
                            If InStr(1, subShp.TextFrame.TextRange.Text, "Question", vbTextCompare) > 0 Then
                                found = True
                                Exit For
                            End If
                        End If
                    End If
                Next j
                ' If found, delete the group shape
                If found Then shp.Delete
            End If
        Next i
    Next sld

    MsgBox "All groups containing 'Question' have been deleted."
End Sub
