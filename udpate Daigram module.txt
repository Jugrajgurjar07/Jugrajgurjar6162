'====================================================================
' FIND ALL EMBEDDED WORD DOCUMENTS
'====================================================================

Private Sub ProcessAllEmbeddedDocuments( _
    ByVal mode As Long, _
    ByVal oldColour As Variant, _
    ByVal newColour As Variant, _
    ByRef totalObjects As Long, _
    ByRef totalChanged As Long)

    Dim shp As Shape
    Dim ils As InlineShape
    Dim childDoc As Document


    '============================================================
    ' FLOATING OLE
    '============================================================

    For Each shp In ActiveDocument.Shapes

        If shp.Type = msoEmbeddedOLEObject Then

            Set childDoc = Nothing

            On Error Resume Next
            Set childDoc = shp.OLEFormat.Object
            On Error GoTo 0

            If Not childDoc Is Nothing Then

                totalObjects = totalObjects + 1

                ProcessDiagramDocument _
                    childDoc, _
                    mode, _
                    oldColour, _
                    newColour, _
                    totalChanged

                On Error Resume Next
                childDoc.Save
                On Error GoTo 0

                CloseChildDocument childDoc

            End If

        End If

    Next shp


    '============================================================
    ' INLINE OLE
    '============================================================

    For Each ils In ActiveDocument.InlineShapes

        If ils.Type = wdInlineShapeEmbeddedOLEObject Then

            Set childDoc = Nothing

            On Error Resume Next
            Set childDoc = ils.OLEFormat.Object
            On Error GoTo 0

            If Not childDoc Is Nothing Then

                totalObjects = totalObjects + 1

                ProcessDiagramDocument _
                    childDoc, _
                    mode, _
                    oldColour, _
                    newColour, _
                    totalChanged

                On Error Resume Next
                childDoc.Save
                On Error GoTo 0

                CloseChildDocument childDoc

            End If

        End If

    Next ils

End Sub


'====================================================================
' PROCESS EMBEDDED WORD DOCUMENT
'====================================================================

Private Sub ProcessDiagramDocument( _
    ByVal doc As Document, _
    ByVal mode As Long, _
    ByVal oldColour As Variant, _
    ByVal newColour As Variant, _
    ByRef totalChanged As Long)

    Dim shp As Shape
    Dim ils As InlineShape
    Dim childDoc As Document


    '============================================================
    ' FLOATING SHAPES
    '============================================================

    For Each shp In doc.Shapes

        ProcessDiagramShape _
            shp, _
            mode, _
            oldColour, _
            newColour, _
            totalChanged

    Next shp


    '============================================================
    ' INLINE OLE INSIDE DIAGRAM
    '============================================================

    For Each ils In doc.InlineShapes

        If ils.Type = wdInlineShapeEmbeddedOLEObject Then

            Set childDoc = Nothing

            On Error Resume Next
            Set childDoc = ils.OLEFormat.Object
            On Error GoTo 0

            If Not childDoc Is Nothing Then

                ProcessDiagramDocument _
                    childDoc, _
                    mode, _
                    oldColour, _
                    newColour, _
                    totalChanged

                On Error Resume Next
                childDoc.Save
                On Error GoTo 0

                CloseChildDocument childDoc

            End If

        End If

    Next ils

End Sub


'====================================================================
' PROCESS EVERY SHAPE
'====================================================================

Private Sub ProcessDiagramShape( _
    ByVal shp As Shape, _
    ByVal mode As Long, _
    ByVal oldColour As Variant, _
    ByVal newColour As Variant, _
    ByRef totalChanged As Long)

    Dim i As Long
    Dim childDoc As Document

    On Error Resume Next


    '============================================================
    ' NESTED EMBEDDED WORD OBJECT
    '============================================================

    If shp.Type = msoEmbeddedOLEObject Then

        Set childDoc = Nothing

        Set childDoc = shp.OLEFormat.Object

        If Not childDoc Is Nothing Then

            ProcessDiagramDocument _
                childDoc, _
                mode, _
                oldColour, _
                newColour, _
                totalChanged

            childDoc.Save

            CloseChildDocument childDoc

        End If

        Set childDoc = Nothing

        Exit Sub

    End If


    '============================================================
    ' GROUP
    '============================================================

    If shp.Type = msoGroup Then

        For i = 1 To shp.GroupItems.Count

            ProcessDiagramShape _
                shp.GroupItems(i), _
                mode, _
                oldColour, _
                newColour, _
                totalChanged

        Next i

    End If


    '============================================================
    ' DRAWING CANVAS
    '============================================================

    If shp.Type = msoCanvas Then

        For i = 1 To shp.CanvasItems.Count

            ProcessDiagramShape _
                shp.CanvasItems(i), _
                mode, _
                oldColour, _
                newColour, _
                totalChanged

        Next i

    End If


    '============================================================
    ' STEP 1
    ' ONLY CANVAS ITEMS
    ' BLACK OUTLINE + TEXT
    '============================================================

    If mode = 1 Then

        ProcessCanvasBlack _
            shp, _
            totalChanged

        Exit Sub

    End If


    '============================================================
    ' STEP 2
    ' BLACK DIAGRAMS + SMALL CIRCLES
    '============================================================

    If mode = 2 Then

        ProcessBlackDiagram _
            shp, _
            totalChanged

        Exit Sub

    End If


    '============================================================
    ' STEP 3-10
    ' COLOUR FIND & REPLACE
    '============================================================

    If mode = 3 Then

        ReplaceDiagramColour _
            shp, _
            CLng(oldColour), _
            CLng(newColour), _
            totalChanged

    End If

    On Error GoTo 0

End Sub


'====================================================================
' STEP 1 PROCESSOR
'====================================================================

Private Sub ProcessCanvasBlack( _
    ByVal shp As Shape, _
    ByRef totalChanged As Long)

    On Error Resume Next


    '============================================================
    ' OUTLINE BLACK
    '============================================================

    Err.Clear

    shp.Line.ForeColor.RGB = RGB(0, 0, 0)

    If Err.Number = 0 Then
        totalChanged = totalChanged + 1
    End If


    '============================================================
    ' TEXT BLACK
    '============================================================

    Err.Clear

    If shp.TextFrame.HasText Then

        shp.TextFrame.TextRange.Font.Color = RGB(0, 0, 0)

        If Err.Number = 0 Then
            totalChanged = totalChanged + 1
        End If

    End If


    '============================================================
    ' TEXTFRAME2 BLACK
    '============================================================

    Err.Clear

    If shp.TextFrame2.HasText Then

        shp.TextFrame2.TextRange.Font.Fill.ForeColor.RGB = _
            RGB(0, 0, 0)

        If Err.Number = 0 Then
            totalChanged = totalChanged + 1
        End If

    End If


    On Error GoTo 0

End Sub


'====================================================================
' STEP 2 PROCESSOR
'====================================================================

Private Sub ProcessBlackDiagram( _
    ByVal shp As Shape, _
    ByRef totalChanged As Long)

    On Error Resume Next


    '============================================================
    ' OUTLINE
    '============================================================

    Err.Clear

    If shp.Line.Visible Then

        shp.Line.ForeColor.RGB = RGB(0, 0, 0)

        If Err.Number = 0 Then
            totalChanged = totalChanged + 1
        End If

    End If


    '============================================================
    ' TEXT
    '============================================================

    Err.Clear

    If shp.TextFrame.HasText Then

        shp.TextFrame.TextRange.Font.Color = RGB(0, 0, 0)

        If Err.Number = 0 Then
            totalChanged = totalChanged + 1
        End If

    End If


    '============================================================
    ' TEXTFRAME2
    '============================================================

    Err.Clear

    If shp.TextFrame2.HasText Then

        shp.TextFrame2.TextRange.Font.Fill.ForeColor.RGB = _
            RGB(0, 0, 0)

        If Err.Number = 0 Then
            totalChanged = totalChanged + 1
        End If

    End If


    '============================================================
    ' SMALL CIRCLE
    '
    ' <= 0.06" x <= 0.06"
    ' approximately circular
    '============================================================

    If IsSmallCircleFinal(shp) Then

        Err.Clear

        shp.Fill.Visible = msoTrue
        shp.Fill.Solid
        shp.Fill.ForeColor.RGB = RGB(0, 0, 0)

        shp.Line.Visible = msoTrue
        shp.Line.ForeColor.RGB = RGB(0, 0, 0)

        If Err.Number = 0 Then
            totalChanged = totalChanged + 1
        End If

    End If


    On Error GoTo 0

End Sub


'====================================================================
' COLOUR FIND & REPLACE
'====================================================================

Private Sub ReplaceDiagramColour( _
    ByVal shp As Shape, _
    ByVal oldRGB As Long, _
    ByVal newRGB As Long, _
    ByRef totalChanged As Long)

    On Error Resume Next


    '============================================================
    ' FILL
    '============================================================

    Err.Clear

    If shp.Fill.Visible Then

        If shp.Fill.ForeColor.RGB = oldRGB Then

            shp.Fill.ForeColor.RGB = newRGB

            If Err.Number = 0 Then
                totalChanged = totalChanged + 1
            End If

        End If

    End If


    '============================================================
    ' OUTLINE
    '============================================================

    Err.Clear

    If shp.Line.Visible Then

        If shp.Line.ForeColor.RGB = oldRGB Then

            shp.Line.ForeColor.RGB = newRGB

            If Err.Number = 0 Then
                totalChanged = totalChanged + 1
            End If

        End If

    End If


    On Error GoTo 0

End Sub


'====================================================================
' SMALL CIRCLE CHECK
'====================================================================

Private Function IsSmallCircleFinal( _
    ByVal shp As Shape) As Boolean

    Dim w As Double
    Dim h As Double
    Dim difference As Double

    On Error GoTo NOT_CIRCLE


    '============================================================
    ' ONLY AUTOSHAPE
    '============================================================

    If shp.Type <> msoAutoShape Then

        IsSmallCircleFinal = False
        Exit Function

    End If


    '============================================================
    ' ONLY OVAL
    '============================================================

    If shp.AutoShapeType <> msoShapeOval Then

        IsSmallCircleFinal = False
        Exit Function

    End If


    '============================================================
    ' 0.06 INCH = 4.32 POINTS
    '============================================================

    w = shp.Width
    h = shp.Height


    If w > 4.32 Then

        IsSmallCircleFinal = False
        Exit Function

    End If


    If h > 4.32 Then

        IsSmallCircleFinal = False
        Exit Function

    End If


    '============================================================
    ' MUST ACTUALLY BE CIRCLE
    '============================================================

    difference = Abs(w - h)

    If difference > 0.15 Then

        IsSmallCircleFinal = False
        Exit Function

    End If


    IsSmallCircleFinal = True

    Exit Function


NOT_CIRCLE:

    IsSmallCircleFinal = False

End Function


'====================================================================
' CLOSE EMBEDDED DOCUMENT
'====================================================================

Private Sub CloseChildDocument( _
    ByVal doc As Document)

    On Error Resume Next

    If Not doc Is ActiveDocument Then

        doc.Close SaveChanges:=wdSaveChanges

    End If

    Err.Clear

    On Error GoTo 0

End Sub
