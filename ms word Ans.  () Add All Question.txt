Sub AddAnswerTagWithTabAndNewLine()
    Dim para As Paragraph
    Dim rng As Range
    Dim answerPos As Long

    ' ???????? ????????? ?? ??? ??? ????
    For Each para In ActiveDocument.Paragraphs
        Set rng = para.Range

        ' "D)" ?? ??? "Ans.    ()" ?????? (??? ?? ?? ?????? ?? ???)
        answerPos = InStr(rng.text, "D)")
        If answerPos > 0 Then
            rng.SetRange Start:=rng.Start + answerPos + 2, End:=rng.End
            rng.InsertAfter "Ans." & vbTab & "()"
            rng.Collapse Direction:=wdCollapseEnd ' ???? ?? ??? ??? ??????? ????
            rng.InsertParagraphAfter ' ?? ?????? ??????
        End If
    Next para
End Sub
